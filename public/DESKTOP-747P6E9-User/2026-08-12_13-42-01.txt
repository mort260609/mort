  (() => {
    const API_BASE = 'https://www.freelancer.com/api';
    const AUTH_HEADER = 'Freelancer-OAuth-V1';
  
    const state = { token: '', userId: null, projects: [], selected: null };
  
    const els = {
      tokenInput: document.getElementById('tokenInput'),
      rememberToken: document.getElementById('rememberToken'),
      connectBtn: document.getElementById('connectBtn'),

    function log(msg, type = 'info') {
      const ts = new Date().toLocaleTimeString('en-GB', { hour12: false });
      const line = document.createElement('div');
      line.className = 'line ' + type;
      line.innerHTML = `<span class="ts">${ts}</span>${escapeHtml(msg)}`;
      els.consoleEl.appendChild(line);
      els.consoleEl.scrollTop = els.consoleEl.scrollHeight;
    }
  
    function escapeHtml(s) {
      return String(s).replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
    }
  
    function money(n, currency) {
      if (n === undefined || n === null) return '—';
      const c = (currency && currency.code) ? currency.code : (currency || 'USD');
      return `${c} ${Number(n).toLocaleString(undefined,{maximumFractionDigits:2})}`;
    }
  
    async function apiCall(path, { method = 'GET', body = null } = {}) {
      if (!state.token) throw new Error('Not connected — paste a token and hit Connect first.');
      const opts = {
        method,
        headers: {
          [AUTH_HEADER]: state.token,
        },
      };
      if (body) {
        opts.headers['Content-Type'] = 'application/json';
        opts.body = JSON.stringify(body);
      }
      let res;
      try {
        res = await fetch(`${API_BASE}${path}`, opts);
      } catch (err) {
        throw new Error(
          `Network/CORS error calling ${path}. The Freelancer API likely blocks direct browser requests — ` +
          `route this call through a small backend proxy that forwards the same headers/body server-side.`
        );
      }
      let json;
      try { json = await res.json(); } catch { json = null; }
      if (!res.ok || (json && json.status === 'error')) {
        const msg = (json && (json.message || JSON.stringify(json.errors))) || `HTTP ${res.status}`;
        throw new Error(`${path} → ${msg}`);
      }
      return json && json.result !== undefined ? json.result : json;
    }
  
    // ---- storage (personal, artifact key/value store) ----
    async function loadRememberedToken() {
      try {
        const r = await window.storage.get('bid-terminal:token', false);
        if (r && r.value) {
          els.tokenInput.value = r.value;
          els.rememberToken.checked = true;
        }
      } catch { /* nothing stored */ }
    }
    async function persistToken(remember) {
      try {
        if (remember) await window.storage.set('bid-terminal:token', els.tokenInput.value, false);
        else await window.storage.delete('bid-terminal:token', false);
      } catch (e) { /* storage optional, ignore failures */ }
    }
  
    // ---- connect / disconnect ----
    async function connect() {
      const token = els.tokenInput.value.trim();
      if (!token) { log('Enter a token first.', 'err'); return; }
      state.token = token;
      els.connectBtn.disabled = true;
      els.connectBtn.textContent = 'Connecting…';
      try {
        const me = await apiCall('/users/0.1/self/?compact=true');
        state.userId = me.id;
        els.statusDot.classList.add('live');
        els.connectBtn.style.display = 'none';
        els.disconnectBtn.style.display = 'inline-block';
        els.tokenInput.disabled = true;
        log(`Connected as ${me.username || me.display_name || ('user #' + me.id)}.`, 'ok');
        await persistToken(els.rememberToken.checked);
      } catch (err) {
        log(err.message, 'err');
        state.token = '';
      } finally {
        els.connectBtn.disabled = false;
        els.connectBtn.textContent = 'Connect';
      }
    }
  
    function disconnect() {
      state.token = '';
      state.userId = null;
      els.statusDot.classList.remove('live');
      els.connectBtn.style.display = 'inline-block';
      els.disconnectBtn.style.display = 'none';
      els.tokenInput.disabled = false;
      log('Disconnected.', 'info');
    }
  
    // ---- search / feed ----
    async function search() {
      const params = new URLSearchParams();
      params.set('compact', 'true');
      params.set('limit', '25');
