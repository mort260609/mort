  class OmniRouteClient:
      def __init__(self, base_url="http://localhost:20128"):
          self.base_url = base_url
          self.api_key = os.getenv("OMNIROUTE_API_KEY")
  
      async def health_check(self, model_combo):
          print(f"[*] 🩺 إرسال فحص نبض سريع (Health Check) للمسار {model_combo}...")
          url = f"{self.base_url}/v1/chat/completions"
          payload = {
              "model": model_combo,
              "messages": [
                  {"role": "user", "content": "Hello, are you receiving this? Reply with 'YES' only."}
              ]
          }
          
          headers = {"Content-Type": "application/json"}
          if self.api_key:
              headers["Authorization"] = f"Bearer {self.api_key}"
  
          timeout = aiohttp.ClientTimeout(total=15) # مهلة قصيرة جداً لاختبار الاستجابة
          
          try:
              async with aiohttp.ClientSession(timeout=timeout) as session:
                  async with session.post(url, json=payload, headers=headers) as response:
                      if response.status == 200:
                          data = await response.json()
                          reply = data['choices'][0]['message']['content']
                          print(f"[+] ✅ نجاح فحص النبض! الرد: {reply.strip()}")
                          return True
                      else:
                          error_text = await response.text()
                          print(f"[-] ❌ فشل فحص النبض (Status {response.status}): {error_text}")
                          return False
          except asyncio.TimeoutError:
              print(f"[-] ❌ تنبيه: انتهت مهلة فحص النبض (Timeout). المسار لا يرد حتى على الطلبات الصغيرة.")
              return False
          except Exception as e:
              print(f"[-] ❌ خطأ غير متوقع أثناء فحص النبض: {e}")
              return False
  
      async def call_agent(self, system_prompt, user_prompt, model_combo):
          url = f"{self.base_url}/v1/chat/completions"
          payload = {
              "model": model_combo,
              "messages": [
                  {"role": "system", "content": system_prompt},
                  {"role": "user", "content": user_prompt}
              ]
          }
          
          headers = {"Content-Type": "application/json"}
          if self.api_key:
              headers["Authorization"] = f"Bearer {self.api_key}"
          else:
              print("[-] تحذير: لم يتم العثور على مفتاح OMNIROUTE_API_KEY في ملف .env")
  
          timeout = aiohttp.ClientTimeout(total=300)
          
          try:
              async with aiohttp.ClientSession(timeout=timeout) as session:
                  async with session.post(url, json=payload, headers=headers) as response:
                      if response.status == 200:
                          data = await response.json()
                          return data['choices'][0]['message']['content']
                      else:
                          error_text = await response.text()
                          print(f"[-] خطأ من OmniRoute (Status {response.status}): {error_text}")
                          return None
          except asyncio.TimeoutError:
              print(f"[-] تنبيه: انتهت مهلة الاتصال (Timeout) للمسار {model_combo} بعد 5 دقائق.")
              return None
          except Exception as e:
              print(f"[-] خطأ غير متوقع في الاتصال: {e}")
              return None
