  ### 🚀 الخطة النهائية لتطوير المنصة
  
  * **المرحلة الأولى: محرك الاستهداف الذكي (Auto-Scoping Engine)**

  class AIRouter:

              ],
              "temperature": 0.2,
              "stream": False
          }

              # رفع مهلة الاتصال إلى 150 ثانية للعقود الضخمة
              async with aiohttp.ClientSession() as session:
                  async with session.post(f"{self.base_url}/chat/completions", headers=self.headers, json=payload, timeout=150) as response:
                      if response.status == 200:
                          data = await response.json()
                          return data['choices'][0]['message']['content']
                      else:
                          error_text = await response.text()
                          print(f"[-] خطأ من OmniRoute (Status {response.status}): {error_text}")
                          return None
          except asyncio.TimeoutError:
              print(f"[-] تنبيه: انتهت مهلة الاتصال (Timeout) للمسار {model_combo}.")
              return None
          except Exception as e:
              print(f"[-] خطأ في الاتصال بالموجه: {e}")
