  class StaticAnalyzer:
      @staticmethod
      async def analyze(contract_path):
          """تشغيل أداة التحليل الثابت الحقيقية (Slither) على ملف العقد"""
          print(f"[*] جاري تشغيل تحليل Slither الثابت على: {contract_path}...")
          
          try:
              # تشغيل أداة Slither داخل دالة منفصلة لتجنب حجب حلقة الحدث الرئيسية (Event Loop)
              def run_slither():
                  result = subprocess.run(
                      ["slither", contract_path],
                      capture_output=True,
                      text=True,
                      encoding="utf-8"
                  )
                  return result.stdout + "\n" + result.stderr
  
              output = await asyncio.to_thread(run_slither)
              
              if output.strip():
                  return f"تقرير Slither الثابت الفعلي:\n{output}"
              else:
                  return "تم فحص العقد بواسطة Slither ولم يتم رصد أي تحذيرات برمجية."
                  
          except FileNotFoundError:
              return "[-] خطأ: أداة Slither غير منصبة أو غير متوفرة في مسار النظام (PATH). يرجى التأكد من تثبيتها بشكل صحيح."
          except Exception as e:
              return f"[-] حدث خطأ غير متوقع أثناء تشغيل Slither: {e}"

          self.test_dir = os.path.join(self.sandbox_dir, "test")

          """تأكد من وجود مجلدات Foundry الأساسية لعدم انهيار النظام"""

          """مقص الكود: استخراج كود Solidity النقي من رد الذكاء الاصطناعي"""

          pattern = r"```solidity\s*(.*?)\s*```"

          return None
