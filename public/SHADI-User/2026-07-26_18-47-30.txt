          """دالة مساعدة لإجراء الطلبات مع 3 محاولات تلقائية (Retry with Exponential Backoff)"""
          for attempt in range(3):
              try:
                  async with aiohttp.ClientSession() as session:
                      async with session.request(method, url, headers=headers, json=json_data, timeout=45) as response:
                          if response.status == 200:
                              return await response.json()
                          elif attempt == 2:
                              return {"error": f"HTTP {response.status}: {await response.text()}"}
              except Exception as e:
                  if attempt == 2:
                      return {"error": str(e)}
              await asyncio.sleep(2 ** attempt) # انتظار 1 ثانية، ثم 2، ثم استسلام

  class AIRouter:
      def __init__(self, base_url="http://localhost:20128/v1"):
          """تهيئة الاتصال بالبوابة المحلية لـ OmniRoute"""
          self.base_url = base_url
          # OmniRoute لا يتطلب مفاتيح للنماذج المجانية، نرسل ترويسة قياسية
          self.headers = {"Content-Type": "application/json", "Authorization": "Bearer sk-omniroute-local"}
  
      async def _safe_request(self, payload):
          """دالة مساعدة لإجراء الطلبات عبر بوابة OmniRoute مع 3 محاولات تلقائية"""
          url = f"{self.base_url}/chat/completions"
          for attempt in range(3):
              try:
                  async with aiohttp.ClientSession() as session:
                      async with session.post(url, headers=self.headers, json=payload, timeout=60) as response:
                          if response.status == 200:
                              return await response.json()
                          elif attempt == 2:
                              return {"error": f"HTTP {response.status}: {await response.text()}"}
              except Exception as e:
                  if attempt == 2:
                      return {"error": str(e)}
              await asyncio.sleep(2 ** attempt)
  
      async def call_kimi_business_logic(self, contract_code):
          """وكيل التدقيق الاقتصادي (يستخدم توجيه auto/smart لأفضل دقة استنتاجية)"""
          prompt = f"أنت مدقق مالي للذكاء الاصطناعي. حلل هذا العقد واستخرج فقط ثغرات المنطق المالي (Business Logic) والتلاعب بالأسعار:\n{contract_code}"
          payload = {"model": "auto/smart", "messages": [{"role": "user", "content": prompt}]}
          
          data = await self._safe_request(payload)
          if "error" not in data and 'choices' in data:
              return data['choices'][0]['message']['content']
          return f"استجابة غير متوقعة/خطأ من OmniRoute: {data}"
