      def __init__(self, base_url="http://localhost:20128/v1", api_key="sk-ce24e9d83b501826-f8735f-feef155d"):
          """الاعتماد الكامل على OmniRoute المحلي لإدارة الـ Combos وحماية الحصص"""

          }
  
      async def call_agent(self, role_prompt, user_prompt, model_combo="auto/smart"):

              "model": model_combo,
              "messages": [
                  {"role": "system", "content": role_prompt},
                  {"role": "user", "content": user_prompt}
              ],
              "temperature": 0.2,
              "stream": False
          }
          try:
              # رفع مهلة الاتصال إلى 150 ثانية للعقود الضخمة
              async with aiohttp.ClientSession() as session:
                  async with session.post(f"{self.base_url}/chat/completions", headers=self.headers, json=payload, timeout=150) as response:
                      if response.status == 200:
                          data = await response.json()
                          return data['choices'][0]['message']['content']
                      else:
                          error_text = await response.text()
                          print(f"[-] خطأ من OmniRoute (Status {response.status}): {error_text}")
                          return None
          except asyncio.TimeoutError:
              print(f"[-] تنبيه: انتهت مهلة الاتصال (Timeout) للمسار {model_combo}.")
              return None
          except Exception as e:
              print(f"[-] خطأ في الاتصال بالموجه: {e}")
