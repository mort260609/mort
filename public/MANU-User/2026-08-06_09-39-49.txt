  rsiSignal= ta.ema(rsiVal, 9)
  atrVal   = ta.atr(atrLen)
  
  // Condiciones de Entrada
  longCondition  = (close > emaTrend) and ta.crossover(rsiVal, rsiSignal) and (rsiVal < 50)
