  CREATE TABLE public.profiles (

      nombre TEXT NOT NULL,
      correo TEXT NOT NULL,
      rol TEXT NOT NULL CHECK (rol IN ('cliente', 'conductor')),

      bio TEXT,
      
      -- Campos exclusivos para conductores
      telefono TEXT,
      licencia TEXT,
      marca_vehiculo TEXT,
      modelo_vehiculo TEXT,
      color_vehiculo TEXT,
      foto_vehiculo TEXT,
      
      created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
  );
  
  -- 3. Crear la tabla de viajes (Trips)
  CREATE TABLE public.trips (
      id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
      origen TEXT NOT NULL,
      destino TEXT NOT NULL,
      fecha_hora TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
      estado TEXT DEFAULT 'pendiente' CHECK (estado IN ('pendiente', 'en_camino', 'finalizado', 'cancelado')),

      conductor_id UUID REFERENCES public.profiles(id) ON DELETE SET NULL,

  -- 1. Limpiar tablas anteriores para garantizar una estructura limpia
  DROP TABLE IF EXISTS public.viajes CASCADE;
  DROP TABLE IF EXISTS public.invitados CASCADE;
  DROP TABLE IF EXISTS public.conductores CASCADE;
  DROP TABLE IF EXISTS public.clientes CASCADE;
  
  -- 2. Tabla: CLIENTES
  -- Los datos de autenticación (correo/contraseña) los maneja Supabase Auth automáticamente.
  CREATE TABLE public.clientes (
      id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
      nombre_completo TEXT NOT NULL,
      correo TEXT NOT NULL,
      bio TEXT,
      created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
  );
  
  -- 3. Tabla: CONDUCTORES
  CREATE TABLE public.conductores (
      id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
      nombre TEXT NOT NULL,
      correo TEXT NOT NULL,
      telefono TEXT NOT NULL,
      numero_licencia TEXT NOT NULL,
      marca_vehiculo TEXT NOT NULL,
      modelo_vehiculo TEXT NOT NULL,
      color_vehiculo TEXT NOT NULL,
      foto_vehiculo TEXT, -- URL de la imagen
      created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
  );
  
  -- 4. Tabla: INVITADOS
  -- Esta tabla guarda a los que piden sin registrarse.
  CREATE TABLE public.invitados (
      id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
      nombre TEXT DEFAULT 'Invitado',
      telefono TEXT NOT NULL,
      created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
  );
  
  -- 5. Tabla: VIAJES
  -- Centraliza todo. El campo 'es_invitado' ayuda a decidir si buscar en 'clientes' o 'invitados'.
  CREATE TABLE public.viajes (
      id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
      origen TEXT NOT NULL,
      destino TEXT NOT NULL,
      fecha_solicitud TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
      estado TEXT DEFAULT 'pendiente' CHECK (estado IN ('pendiente', 'en_camino', 'finalizado', 'cancelado')),
      
      -- Relación: Si es NULL, se revisa la columna de invitados o es invitado
      cliente_id UUID REFERENCES public.clientes(id) ON DELETE SET NULL,
      invitado_id UUID REFERENCES public.invitados(id) ON DELETE SET NULL,
      
      -- Quién tomó el viaje
      conductor_id UUID REFERENCES public.conductores(id) ON DELETE SET NULL,
      
      created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
  );
  
  -- Índices para que la app no se ponga lenta al cargar muchos viajes
  CREATE INDEX idx_viajes_fecha ON public.viajes (fecha_solicitud DESC);
  CREATE INDEX idx_viajes_estado ON public.viajes (estado);

  # TÉRMINOS Y CONDICIONES DE USO
  
  ## ASERE TAXIS
  
  **Última actualización:** __________________
  
  Estos Términos y Condiciones regulan el acceso y uso de la plataforma tecnológica **Asere Taxis** y de sus servicios relacionados.
  
  Al registrarse, acceder o utilizar la aplicación, el usuario declara que ha leído, comprendido y aceptado estos Términos y Condiciones.
  
  Si el usuario no está de acuerdo con alguna de sus disposiciones, deberá abstenerse de utilizar la aplicación.
  
  ---
  
  # 1. IDENTIFICACIÓN DE LA PLATAFORMA
  
  **Asere Taxis** es una plataforma tecnológica destinada a facilitar la conexión entre personas que solicitan servicios de transporte privado y conductores afiliados que ofrecen dichos servicios mediante sus propios vehículos.
  
  La plataforma puede incluir aplicaciones móviles, sitios web, sistemas administrativos y otros medios tecnológicos relacionados con la operación de Asere Taxis.
  
  ---
  
  # 2. ACEPTACIÓN DE LOS TÉRMINOS
  
  El registro y uso de la plataforma implica la aceptación expresa de estos Términos y Condiciones.
  
  El usuario deberá proporcionar información verdadera, completa y actualizada durante el proceso de registro.
  
  Asere Taxis podrá solicitar información adicional cuando resulte necesaria para verificar la identidad, seguridad o elegibilidad del usuario.
  
  ---
  
  # 3. CUENTAS DE USUARIO
  
  Cada persona deberá utilizar una cuenta individual.
  
  El usuario es responsable de:
  
  * Mantener la confidencialidad de sus credenciales.
  * No compartir su cuenta.
  * Proporcionar información correcta.
  * Informar inmediatamente cualquier acceso no autorizado.
  * Utilizar la plataforma únicamente para fines legítimos.
  
  Asere Taxis podrá suspender o cancelar cuentas cuando existan indicios de fraude, abuso, información falsa, conducta inapropiada o incumplimiento de estos Términos.
  
  ---
  
  # 4. SERVICIO DE TRANSPORTE
  
  Asere Taxis facilita la conexión entre pasajeros y conductores afiliados.
  
  La disponibilidad de un viaje dependerá de los conductores disponibles en el momento de la solicitud.
  
  Las características del servicio, incluyendo disponibilidad, tarifa estimada, ubicación y tiempo aproximado de llegada, podrán variar según las condiciones operativas.
  
  ---
  
  # 5. CONDUCTORES INDEPENDIENTES
  
  Los conductores que utilizan Asere Taxis prestan sus servicios de manera independiente.
  
  Cada conductor es responsable de cumplir con los requisitos legales aplicables para conducir y transportar pasajeros, así como de mantener en condiciones adecuadas el vehículo utilizado.
  
  La afiliación a Asere Taxis no convierte al conductor en empleado, socio, accionista o representante legal de la empresa.
  
  Los conductores deberán aceptar y cumplir adicionalmente el **Contrato de Prestación de Servicios para Conductores Afiliados** y las normas operativas establecidas por Asere Taxis.
  
  ---
  
  # 6. REQUISITOS DE LOS CONDUCTORES
  
  Antes de ser autorizado para prestar servicios mediante la plataforma, el conductor podrá estar sujeto a procesos de verificación establecidos por Asere Taxis.
  
  Estos procesos podrán incluir la revisión de:
  
  * Identidad.
  * Licencia de conducir.
  * Documentación del vehículo.
  * Seguro correspondiente.
  * Condiciones del vehículo.
  * Información personal requerida para la operación.
  
  Asere Taxis podrá rechazar, suspender o retirar la autorización de un conductor cuando no cumpla los requisitos establecidos.
  
  ---
  
  # 7. SOLICITUD Y ACEPTACIÓN DE VIAJES
  
  Los pasajeros podrán solicitar viajes mediante la aplicación.
  
  El conductor podrá aceptar o rechazar solicitudes de acuerdo con las funciones disponibles en la plataforma.
  
  Una vez aceptado un viaje, ambas partes deberán actuar de manera responsable y respetuosa.
  
  ---
  
  # 8. CANCELACIONES
  
  Un viaje podrá ser cancelado por el pasajero o por el conductor bajo las condiciones establecidas por Asere Taxis.
  
  Cuando corresponda, Asere Taxis podrá establecer cargos o consecuencias por cancelaciones injustificadas, reiteradas o abusivas.
  
  Las condiciones específicas podrán ser modificadas y comunicadas mediante la aplicación.
  
  ---
  
  # 9. TARIFAS Y PAGOS
  
  Las tarifas serán determinadas de acuerdo con el modelo comercial vigente de Asere Taxis.
  
  Antes o durante la solicitud del servicio, la aplicación podrá mostrar al usuario información relacionada con el precio estimado o aplicable.
  
  El usuario acepta pagar el importe correspondiente al servicio contratado mediante los métodos de pago disponibles.
  
  Asere Taxis podrá modificar tarifas, métodos de pago, promociones o condiciones comerciales cuando resulte necesario para la operación del servicio.
  
  ---
  
  # 10. PROMOCIONES Y DESCUENTOS
  
  Asere Taxis podrá ofrecer promociones, descuentos, créditos u otros beneficios.
  
  Cada promoción podrá estar sujeta a condiciones específicas.
  
  Las promociones no podrán utilizarse mediante fraude, manipulación o creación de cuentas destinadas exclusivamente a obtener beneficios indebidos.
  
  ---
  
  # 11. CONDUCTA DE LOS USUARIOS
  
  Todos los usuarios deberán mantener una conducta respetuosa.
  
  Está prohibido utilizar la plataforma para:
  
  * Amenazar, acosar o agredir a otras personas.
  * Cometer actividades ilegales.
  * Dañar intencionalmente un vehículo.
  * Utilizar información falsa.
  * Realizar actividades fraudulentas.
  * Intentar manipular la plataforma.
  * Utilizar la cuenta de otra persona.
  * Realizar cualquier actividad que pueda poner en peligro a otros usuarios o conductores.
  
  Asere Taxis podrá suspender o cancelar cuentas que incumplan estas normas.
  
  ---
  
  # 12. SEGURIDAD
  
  La seguridad de pasajeros y conductores constituye una prioridad de Asere Taxis.
  
  Los usuarios deberán utilizar los mecanismos de seguridad disponibles dentro de la aplicación y reportar inmediatamente cualquier situación que represente un riesgo.
  
  En situaciones de emergencia, el usuario deberá contactar a los servicios de emergencia correspondientes.
  
  Asere Taxis podrá colaborar con las autoridades competentes cuando sea legalmente requerido.
  
  ---
  
  # 13. OBJETOS PERDIDOS
  
  Asere Taxis podrá facilitar la comunicación entre pasajero y conductor cuando se reporte un objeto perdido.
  
  La empresa no garantiza la recuperación de objetos olvidados dentro de los vehículos.
  
  Los usuarios deberán informar cualquier objeto perdido lo antes posible mediante los canales oficiales disponibles.
  
  ---
  
  # 14. DAÑOS AL VEHÍCULO
  
  El pasajero será responsable por daños que cause intencionalmente o por conducta negligente al vehículo del conductor.
  
  Cuando corresponda, Asere Taxis podrá facilitar la comunicación entre las partes para gestionar el incidente.
  
  ---
  
  # 15. INFORMACIÓN Y DATOS PERSONALES
  
  Para proporcionar sus servicios, Asere Taxis podrá recopilar y procesar información proporcionada por los usuarios, incluyendo información de identificación, contacto, ubicación y datos relacionados con el uso de la plataforma.
  
  La información será utilizada para fines relacionados con:
  
  * Creación y administración de cuentas.
  * Prestación del servicio.
  * Seguridad.
  * Comunicación.
  * Procesamiento de pagos.
  * Atención al cliente.
  * Prevención de fraude.
  * Mejora de la plataforma.
  * Cumplimiento de obligaciones legales.
  
  El tratamiento de datos personales estará sujeto a la legislación aplicable en la República de Surinam y a la Política de Privacidad de Asere Taxis.
  
  ---
  
  # 16. UBICACIÓN
  
  Para proporcionar determinadas funciones, la aplicación podrá solicitar acceso a la ubicación del dispositivo del usuario.
  
  La información de ubicación podrá utilizarse para:
  
  * Solicitar y asignar viajes.
  * Mostrar la ubicación del conductor.
  * Facilitar la navegación.
  * Calcular distancias y tiempos.
  * Mejorar la seguridad y operación del servicio.
  
  El usuario podrá administrar los permisos de ubicación desde la configuración de su dispositivo, aunque determinadas funciones podrían dejar de estar disponibles.
  
  ---
  
  # 17. PROPIEDAD INTELECTUAL
  
  La aplicación, marca, logotipos, diseños, interfaces, contenidos, software, bases de datos y demás elementos tecnológicos de Asere Taxis son propiedad de la empresa o se utilizan bajo las autorizaciones correspondientes.
  
  El uso de la aplicación no concede al usuario ningún derecho de propiedad sobre dichos elementos.
  
  Queda prohibida la copia, modificación, distribución, ingeniería inversa, reproducción o explotación no autorizada de cualquier parte de la plataforma.
  
  ---
  
  # 18. DISPONIBILIDAD DE LA PLATAFORMA
  
  Asere Taxis procurará mantener la plataforma disponible y funcionando correctamente.
  
  Sin embargo, podrán producirse interrupciones por:
  
  * Mantenimiento.
  * Actualizaciones.
  * Fallos técnicos.
  * Problemas de conectividad.
  * Servicios de terceros.
  * Situaciones de fuerza mayor.
  * Circunstancias fuera del control razonable de Asere Taxis.
  
  La empresa no garantiza que la plataforma estará disponible de manera ininterrumpida en todo momento.
  
  ---
  
  # 19. SERVICIOS DE TERCEROS
  
  La plataforma podrá utilizar servicios tecnológicos proporcionados por terceros, incluyendo servicios de mapas, geolocalización, procesamiento de pagos, almacenamiento, comunicaciones y otros servicios necesarios para su funcionamiento.
  
  La disponibilidad de determinadas funciones podrá depender de dichos proveedores.
  
  ---
  
  # 20. RESPONSABILIDAD
  
  Asere Taxis actúa como plataforma tecnológica de intermediación.
  
  Los conductores son responsables de cumplir las obligaciones legales y operativas correspondientes a la prestación del servicio de transporte.
  
  Nada de lo establecido en estos Términos pretende excluir o limitar responsabilidades que legalmente no puedan ser excluidas o limitadas.
  
  Los usuarios deberán actuar de manera responsable y cumplir las leyes aplicables.
  
  ---
  
  # 21. MODIFICACIONES
  
  Asere Taxis podrá modificar estos Términos y Condiciones cuando resulte necesario debido a cambios en el servicio, legislación, tecnología o modelo de negocio.
  
  Cuando corresponda, los usuarios serán informados de modificaciones relevantes mediante la aplicación u otros canales disponibles.
  
  El uso continuado de la plataforma después de la entrada en vigor de los cambios podrá constituir aceptación de los nuevos términos, en la medida permitida por la legislación aplicable.
  
  ---
  
  # 22. SUSPENSIÓN Y TERMINACIÓN DE CUENTAS
  
  Asere Taxis podrá suspender o cancelar una cuenta cuando exista incumplimiento de estos Términos, fraude, conducta abusiva, información falsa, riesgo para la seguridad o cualquier otra circunstancia que justifique razonablemente la medida.
  
  El usuario también podrá solicitar la cancelación de su cuenta mediante los mecanismos disponibles.

  ---

  * Confirmaciones de viajes.

  * Actualizaciones de cuenta.
  * Información administrativa.
  * Comunicaciones relacionadas con pagos.
  
  Las comunicaciones promocionales estarán sujetas a las opciones de consentimiento y a la legislación aplicable.
  
  ---
  
  # 24. LEGISLACIÓN APLICABLE
  
  Estos Términos y Condiciones se regirán por las leyes aplicables de la **República de Surinam**.
  
  Cualquier controversia relacionada con el uso de la plataforma será tratada inicialmente mediante comunicación directa entre las partes, sin perjuicio del derecho de cualquiera de ellas de acudir a las autoridades o tribunales competentes cuando corresponda.
