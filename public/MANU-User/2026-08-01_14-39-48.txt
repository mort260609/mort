  function performAttack() {
    if (stats.stamina < stats.attackCost) return;
    stats.stamina -= stats.attackCost;
    attackAnimTime = 0.4;
  
    // Detectar enemigos en rango
    const attackRange = 2.5;
    const forward = new THREE.Vector3(0, 0, 1).applyAxisAngle(new THREE.Vector3(0,1,0), cameraYaw);
    
    for (const enemy of enemies) {
      const toEnemy = enemy.position.clone().sub(player.position);
      toEnemy.y = 0;
      const dist = toEnemy.length();

        const dot = toEnemy.normalize().dot(forward);
        if (dot > 0.3) {
          enemy.userData.health -= 35;
          // Efecto de golpe
          const flash = new THREE.Mesh(
            new THREE.SphereGeometry(0.5, 8, 8),

            scene.remove(enemy);

            setTimeout(spawnEnemy, 2000);
          }
        }
      }
    }
  }

  const clock = new THREE.Clock();
  
  function update() {
    const dt = Math.min(clock.getDelta(), 0.05);
  
    // Movimiento del jugador
    const moveSpeed = 6;
    let moveX = 0, moveZ = 0;
    if (keys['KeyW']) moveZ += 1;
    if (keys['KeyS']) moveZ -= 1;
    if (keys['KeyA']) moveX -= 1;
    if (keys['KeyD']) moveX += 1;
  
    const moving = moveX !== 0 || moveZ !== 0;
    
    if (moving) {
      const dir = new THREE.Vector3(moveX, 0, moveZ).normalize();
      dir.applyAxisAngle(new THREE.Vector3(0,1,0), cameraYaw);
      player.position.x += dir.x * moveSpeed * dt;
      player.position.z += dir.z * moveSpeed * dt;
      
      // Rotar personaje hacia dirección de movimiento
      const targetRot = Math.atan2(dir.x, dir.z);
      player.rotation.y = targetRot;
      
      walkTime += dt * 10;
      player.userData.legL.rotation.x = Math.sin(walkTime) * 0.6;
      player.userData.legR.rotation.x = -Math.sin(walkTime) * 0.6;
      player.userData.armR.rotation.x = -Math.sin(walkTime) * 0.4;
    } else {
      player.userData.legL.rotation.x *= 0.8;
      player.userData.legR.rotation.x *= 0.8;
      player.userData.armR.rotation.x *= 0.8;
    }
  
    // Salto
    if (keys['Space'] && isGrounded) {
      playerVel.y = 9;
      isGrounded = false;
    }
    playerVel.y += gravity * dt;
    player.position.y += playerVel.y * dt;
    if (player.position.y <= 0) {
      player.position.y = 0;
      playerVel.y = 0;
      isGrounded = true;
    }
  
    // Limitar a la arena
    const distFromCenter = Math.sqrt(player.position.x**2 + player.position.z**2);
    if (distFromCenter > 23) {
      const ang = Math.atan2(player.position.z, player.position.x);
      player.position.x = Math.cos(ang) * 23;
      player.position.z = Math.sin(ang) * 23;
    }
  
    // Animación de ataque
    if (attackAnimTime > 0) {
      attackAnimTime -= dt;
      const t = 1 - (attackAnimTime / 0.4);
      player.userData.sword.rotation.z = -0.3 - Math.sin(t * Math.PI) * 2;
      player.userData.sword.rotation.x = Math.sin(t * Math.PI) * 1.5;
    } else {
      player.userData.sword.rotation.z = -0.3;
      player.userData.sword.rotation.x = 0;
    }
  
    // Cámara 3ra persona
    const camDist = 5;
    const camOffset = new THREE.Vector3(
      Math.sin(cameraYaw) * Math.cos(cameraPitch) * camDist,
      Math.sin(cameraPitch) * camDist + 2,
      Math.cos(cameraYaw) * Math.cos(cameraPitch) * camDist
    );
    const targetCamPos = player.position.clone().add(camOffset);
    camera.position.lerp(targetCamPos, 0.15);
    camera.lookAt(player.position.x, player.position.y + 1.5, player.position.z);
  
    // Estamina
    if (stats.stamina < stats.maxStamina) {
      stats.stamina = Math.min(stats.maxStamina, stats.stamina + stats.staminaRegen * dt);
    }
  
    // Enemigos: movimiento simple hacia el jugador
    for (const enemy of enemies) {
      const toPlayer = player.position.clone().sub(enemy.position);
      toPlayer.y = 0;
      const d = toPlayer.length();
      if (d > 1.5) {
        toPlayer.normalize();
        enemy.position.x += toPlayer.x * 1.5 * dt;
        enemy.position.z += toPlayer.z * 1.5 * dt;
      }
      // Rebote
      enemy.userData.baseY += dt * 3;
      enemy.children[0].scale.y = 0.75 + Math.sin(enemy.userData.baseY) * 0.05;
      
      // Daño al jugador si está cerca
      if (d < 1.5 && Math.random() < dt * 0.5) {
        stats.health = Math.max(0, stats.health - 5);
        document.getElementById('damageFlash').style.opacity = '1';
        setTimeout(() => document.getElementById('damageFlash').style.opacity = '0', 200);
        if (stats.health <= 0) {
          stats.health = stats.maxHealth;
          player.position.set(0,0,0);
        }
      }
    }
  
    // UI
    document.getElementById('healthFill').style.width = (stats.health/stats.maxHealth*100) + '%';
    document.getElementById('staminaFill').style.width = (stats.stamina/stats.maxStamina*100) + '%';
  }
