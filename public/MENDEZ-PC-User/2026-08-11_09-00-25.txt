  // SPDX-License-Identifier: MIT
  pragma solidity ^0.8.20;
  
  import {Script, console} from "forge-std/Script.sol";
  import {TimelockController} from "@openzeppelin/contracts/governance/TimelockController.sol";
  
  /**
   * @title DeployTimelock
   * @dev Step 7b of the reviewed NEXAI_Redeployment_Sequence.pdf. Deploys OpenZeppelin's
   * TimelockController wired per NEXAI_Multisig_Timelock_Design.pdf's 48h delay, pointed
   * at the REAL nested Outer AdminSafe from step 7a (DeployNestedAdminSafe.s.sol -
   * Mendez's wallet + Dmitry's 1-of-2 inner Safe, 2-of-2), not the old flat 2-of-3
   * placeholder Safe this file originally referenced (DeploySafes.s.sol's provisional
   * signer list, never actually deployed - see step 7a's doc comment).
   *
   *  - 48h minimum delay on all governance changes
   *  - Outer AdminSafe is the sole proposer - OZ's constructor also auto-grants
   *    proposers the canceller role, so the Outer AdminSafe (2-of-2: Mendez + inner
   *    Safe) can already cancel a queued proposal through the normal admin path
   *  - Execution is permissionless (executors = [address(0)]): once a proposal clears
   *    the delay, anyone can execute exactly what was proposed and approved
   *
   * FLAGGED, NOT GUESSED: NEXAI_Multisig_Timelock_Design.pdf calls for a separate 1-of-3
   * guardian threshold able to cancel a queued proposal independently of pause and
   * without needing the full admin quorum - a fast, low-friction safety valve distinct
   * from the Outer AdminSafe's own (slower, 2-of-2-gated) cancel ability above. No
   * GuardianSafe exists yet (out of scope for this step per the sequence doc), so this
   * script deliberately does NOT grant CANCELLER_ROLE to anything beyond what the
   * Outer AdminSafe already gets automatically as proposer. Granting a real
   * fast-path CANCELLER_ROLE is follow-up work once a GuardianSafe structure is
   * actually decided/scoped - not assumed here to be "same as the Outer AdminSafe" or
   * skipped silently.
   *
   * The deployer is set as a temporary TIMELOCK_ADMIN_ROLE holder purely to grant the
   * proposer role, then immediately renounces - matching OZ's own documented pattern
   * ("should be subsequently renounced in favor of administration through timelocked
   * proposals"). From that point on the timelock is self-administered.
   * Testnet only.
   */
  contract DeployTimelockScript is Script {
      uint256 constant MIN_DELAY = 48 hours;
  
      // Real Outer AdminSafe from step 7a (DeployNestedAdminSafe.s.sol) - not a
      // provisional placeholder.
      address constant ADMIN_SAFE = 0xEe8b34C12A3946CD14b498C5056525F7404F34eC;

      function run() external {
          require(ADMIN_SAFE != address(0), "Set ADMIN_SAFE to DeployNestedAdminSafe.s.sol's Outer AdminSafe address first");
  
          uint256 deployerPrivateKey = vm.envUint("PRIVATE_KEY");
          address deployer = vm.addr(deployerPrivateKey);
          vm.startBroadcast(deployerPrivateKey);
  
          address[] memory proposers = new address[](1);
          proposers[0] = ADMIN_SAFE;
  
          address[] memory executors = new address[](1);
          executors[0] = address(0); // open/permissionless executor role
  
          TimelockController timelock = new TimelockController(MIN_DELAY, proposers, executors, deployer);
          console.log("TimelockController deployed at:", address(timelock));
          console.log("Proposer (Outer AdminSafe):", ADMIN_SAFE);
          console.log("No separate guardian CANCELLER_ROLE granted - GuardianSafe not yet scoped, see doc comment.");
  
          timelock.renounceRole(timelock.TIMELOCK_ADMIN_ROLE(), deployer);
          console.log("Renounced deployer's TIMELOCK_ADMIN_ROLE - timelock is now self-administered");
  
          vm.stopBroadcast();
      }
  }

  /**
   * @title DeployTimelock
   * @dev Step 7b of the reviewed NEXAI_Redeployment_Sequence.pdf. Deploys OpenZeppelin's
   * TimelockController wired per NEXAI_Multisig_Timelock_Design.pdf's 48h delay, pointed
   * at the REAL nested Outer AdminSafe from step 7a (DeployNestedAdminSafe.s.sol -
   * Mendez's wallet + Dmitry's 1-of-2 inner Safe, 2-of-2), not the old flat 2-of-3

   *  - Outer AdminSafe is the sole proposer - OZ's constructor also auto-grants
   *    proposers the canceller role, so the Outer AdminSafe (2-of-2: Mendez + inner
   *    Safe) can already cancel a queued proposal through the normal admin path
   *  - Execution is permissionless (executors = [address(0)]): once a proposal clears
   *    the delay, anyone can execute exactly what was proposed and approved
   *
   * FLAGGED, NOT GUESSED: NEXAI_Multisig_Timelock_Design.pdf calls for a separate 1-of-3
   * guardian threshold able to cancel a queued proposal independently of pause and
   * without needing the full admin quorum - a fast, low-friction safety valve distinct
   * from the Outer AdminSafe's own (slower, 2-of-2-gated) cancel ability above. No
   * GuardianSafe exists yet (out of scope for this step per the sequence doc), so this
   * script deliberately does NOT grant CANCELLER_ROLE to anything beyond what the
   * Outer AdminSafe already gets automatically as proposer. Granting a real
   * fast-path CANCELLER_ROLE is follow-up work once a GuardianSafe structure is
   * actually decided/scoped - not assumed here to be "same as the Outer AdminSafe" or
   * skipped silently.
   *
   * The deployer is set as a temporary TIMELOCK_ADMIN_ROLE holder purely to grant the
   * proposer role, then immediately renounces - matching OZ's own documented pattern
   * ("should be subsequently renounced in favor of administration through timelocked
   * proposals"). From that point on the timelock is self-administered.
   * Testnet only.
   */
  contract DeployTimelockScript is Script {
      uint256 constant MIN_DELAY = 48 hours;
  
      // Real Outer AdminSafe from step 7a (DeployNestedAdminSafe.s.sol) - not a
      // provisional placeholder.
